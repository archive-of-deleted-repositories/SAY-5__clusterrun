find_package(GTest QUIET)
if(NOT GTest_FOUND)
  find_package(PkgConfig QUIET)
  find_library(GTEST_LIB gtest)
  find_library(GTEST_MAIN_LIB gtest_main)
endif()

function(clusterrun_add_test name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE clusterrun_core clusterrun_transport
                        clusterrun_controller_lib clusterrun_worker_lib)
  if(GTest_FOUND)
    target_link_libraries(${name} PRIVATE GTest::gtest GTest::gtest_main)
  else()
    target_link_libraries(${name} PRIVATE ${GTEST_LIB} ${GTEST_MAIN_LIB} pthread)
  endif()
  add_test(NAME ${name} COMMAND ${name})
endfunction()
